\begin{frame}{Content}
\begin{itemize} 
	\item  5-10 minutes presentation
	\vspace{0.2cm}
	\begin{itemize} 
		\item My suggestion: around 5-7 slides
		\vspace{0.2cm}
	\end{itemize}
	\item To control time, audiences can only ask questions after presentation (2-minutes Q\&A).
	\vspace{0.2cm}
	
\end{itemize}
\end{frame}




\begin{frame}{Content}{Page 1}
	\begin{itemize} 
		\item Introduce your research topic 
		\vspace{0.2cm}
		\begin{itemize}
			\item What is the causal relationship you want to estimate
			\vspace{0.2cm}
			\item Briefly discuss why it is interesting
	\end{itemize}
	\end{itemize}
\end{frame}







\begin{frame}{Content}{Page 2}
	\begin{itemize} 
		\item Discuss your data and sample
		\vspace{0.2cm}
		\begin{itemize}
			\item Introduce the datasets you use
			\vspace{0.2cm} 
			\item Discuss how your sample selection criteria
			\vspace{0.2cm} 	
			\begin{itemize}
				\item Sample period
				\vspace{0.2cm}
				\item Impose any restrictions on the estimation sample
			\end{itemize}
		\end{itemize}
	\end{itemize}
\end{frame}




\begin{frame}{Content}{Page 3}
	\begin{itemize} 
		\item Discuss your empirical method
		\vspace{0.2cm}
		\begin{itemize}
			\item Write down your estimated equations
			\vspace{0.2cm} 
			\item Define the variable and notations used in your estimation equation
			\vspace{0.2cm} 	
			\begin{itemize}
				\item The outcome variable
				\vspace{0.2cm}
				\item The treatment variable
				\vspace{0.2cm} 
				\item Important covariates or fixed effects
			\end{itemize}
		\end{itemize}
	\end{itemize}
\end{frame}



\begin{frame}{Content}{Page 4 - 5}
	\begin{itemize} 
		\item Show and discuss summary statistics of your sample
		\vspace{0.2cm}
	    \item Present the preliminary results
	    		\vspace{0.2cm}
	    	\begin{itemize} 
	    	\item If possible, present the preliminary results
	    \vspace{0.2cm} 
	    	    \item Use tables or figures		
	    	    	    	    \vspace{0.2cm} 	
	    \item Interpret your estimates	


	    			\end{itemize}
    		\vspace{0.2cm}
    	\item Finally, you can discuss what will you do before you submitting your term paper (future work)		
	\end{itemize}
\end{frame}
